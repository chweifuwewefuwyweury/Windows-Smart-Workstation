# ================= 2. 物理减速缓动算法 =================
def ease_out_quint(t):
    return 1 - math.pow(1 - t, 5)


# ================= 3. Win11 现代渐变背景画布 =================
class FluentBackground(tk.Canvas):
    def __init__(self, parent, is_dark=True, **kwargs):
        super().__init__(parent, highlightthickness=0, bd=0, **kwargs)
        self.is_dark = is_dark
        self.bind("<Configure>", self._draw_gradient)

    def set_mode(self, is_dark):
        self.is_dark = is_dark
        self._draw_gradient()

    def _draw_gradient(self, event=None):
        self.delete("all")
        w, h = self.winfo_width(), self.winfo_height()
        if w <= 0 or h <= 0:
            return

        if self.is_dark:
            c1, c2 = (24, 28, 36), (15, 17, 23)
        else:
            c1, c2 = (240, 244, 250), (225, 232, 242)

        steps = 60
        for i in range(steps):
            r = int(c1[0] + (c2[0] - c1[0]) * (i / steps))
            g = int(c1[1] + (c2[1] - c1[1]) * (i / steps))
            b = int(c1[2] + (c2[2] - c1[2]) * (i / steps))
            color = f"#{r:02x}{g:02x}{b:02x}"
            y1 = int(h * (i / steps))
            y2 = int(h * ((i + 1) / steps))
            self.create_rectangle(0, y1, w, y2, fill=color, outline=color)


# ================= 4. Win11 现代圆角阴影卡片 =================
class FluentCard(tk.Canvas):
    def __init__(self, parent, radius=12, bg_color="#272727", border_color="#3a3a3a", **kwargs):
        parent_bg = "#181c24"
        try:
            if isinstance(parent, dict) or hasattr(parent, "__getitem__"):
                p_bg = parent["bg"]
                if p_bg:
                    parent_bg = p_bg
        except Exception:
            pass

        super().__init__(parent, bg=parent_bg, highlightthickness=0, bd=0, **kwargs)
        self.radius = radius
        self.bg_color = bg_color
        self.border_color = border_color
        self.bind("<Configure>", self._redraw)

    def set_colors(self, bg_color, border_color):
        self.bg_color = bg_color
        self.border_color = border_color
        if self.master:
            try:
                p_bg = self.master["bg"]
                if p_bg:
                    self.config(bg=p_bg)
            except Exception:
                pass
        self._redraw()

    def _redraw(self, event=None):
        self.delete("all")
        w, h = self.winfo_width(), self.winfo_height()
        r = min(self.radius, w // 2, h // 2)

        if r > 0 and w > 0 and h > 0:
            self.create_arc(0, 0, 2 * r, 2 * r, start=90, extent=90, fill=self.border_color, outline=self.border_color)
            self.create_arc(w - 2 * r, 0, w, 2 * r, start=0, extent=90, fill=self.border_color,
                            outline=self.border_color)
            self.create_arc(w - 2 * r, h - 2 * r, w, h, start=270, extent=90, fill=self.border_color,
                            outline=self.border_color)
            self.create_arc(0, h - 2 * r, 2 * r, h, start=180, extent=90, fill=self.border_color,
                            outline=self.border_color)
            self.create_rectangle(r, 0, w - r, h, fill=self.border_color, outline=self.border_color)
            self.create_rectangle(0, r, w, h - r, fill=self.border_color, outline=self.border_color)

            self.create_arc(1, 1, 2 * r - 1, 2 * r - 1, start=90, extent=90, fill=self.bg_color, outline=self.bg_color)
            self.create_arc(w - 2 * r + 1, 1, w - 1, 2 * r - 1, start=0, extent=90, fill=self.bg_color,
                            outline=self.bg_color)
            self.create_arc(w - 2 * r + 1, h - 2 * r + 1, w - 1, h - 1, start=270, extent=90, fill=self.bg_color,
                            outline=self.bg_color)
            self.create_arc(1, h - 2 * r + 1, 2 * r - 1, h - 1, start=180, extent=90, fill=self.bg_color,
                            outline=self.bg_color)
            self.create_rectangle(r, 1, w - r, h - 1, fill=self.bg_color, outline=self.bg_color)
            self.create_rectangle(1, r, w - 1, h - r, fill=self.bg_color, outline=self.bg_color)

WEATHER_DICT = {
    "sunny": "晴朗", "clear": "晴朗", "partly cloudy": "多云", "cloudy": "阴天", "overcast": "阴天",
    "mist": "薄雾", "fog": "大雾", "haze": "霾", "smoky haze": "烟雾/霾", "smoke": "烟雾",
    "freezing fog": "冻雾", "patchy rain possible": "局部有雨", "patchy snow possible": "局部有雪",
    "thundery outbreaks possible": "可能有雷阵雨", "blowing snow": "吹雪", "blizzard": "暴风雪",
    "drizzle": "细雨/毛毛雨", "light rain": "小雨", "moderate rain": "中雨", "heavy rain": "大雨",
    "light snow": "小雪", "moderate snow": "中雪", "heavy snow": "大雪", "ice pellets": "冰雹",
    "thunderstorm": "雷暴", "rain shower": "阵雨", "snow shower": "阵雪"
}


def translate_weather(text):
    if not text:
        return "未知"
    lower_text = text.strip().lower()
    if lower_text in WEATHER_DICT:
        return WEATHER_DICT[lower_text]
    for en, zh in WEATHER_DICT.items():
        if en in lower_text:
            return zh
    return text

    def _auto_detect_location(self):
        try:
            url = "http://ip-api.com/json/?lang=zh-CN"
            req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
            with urllib.request.urlopen(req, timeout=5) as resp:
                data = json.loads(resp.read().decode('utf-8'))

                self.current_location_zh = city
        except Exception:
            self.current_location_zh = "北京"

            frame.config(bg=self.colors["menu_bg"])

    def toggle_menu(self):
        if self.menu_expanded:
            self.menu_dropdown.place_forget()
            self.menu_expanded = False
        else:
            self.menu_dropdown.config(bg=self.colors["menu_bg"], highlightbackground=self.colors["menu_border"])
            self._refresh_menu_items_theme()
            self.menu_dropdown.place(x=20, y=72, width=320)
            self.menu_dropdown.lift()
            self.menu_expanded = True

    def _on_global_click(self, event):
        if not self.menu_expanded:
            return

        w = event.widget
        while w:
            if w in (self.menu_dropdown, self.btn_menu):
                return
            w = getattr(w, "master", None)

        self.toggle_menu()

    def _on_menu_item_click(self, page_name):
        if self.menu_expanded:
            self.toggle_menu()
        self.switch_page(page_name)

    # ================= 7. 页面切换动画 =================
    def switch_page(self, page_name):
        self.current_page_name = page_name
        self.lbl_title.config(text=page_name)

        if self.current_frame:
            self.current_frame.destroy()

        new_frame = tk.Frame(self.main_area, bg=self.colors["window_bg"])
        self.current_frame = new_frame

        if page_name == "主页":
            self._build_home_dashboard(new_frame)
        elif page_name == "文件重命名":
            self._build_file_renamer(new_frame)
        elif page_name == "待办清单":
            self._build_todo_list(new_frame)
        elif page_name == "剪贴板历史":
            self._build_clipboard_history(new_frame)
        elif page_name == "计算器":
            self._build_calculator(new_frame)
        elif page_name == "天气查询":
            self._build_weather_query(new_frame)

        start_y = 15
        duration = 10
        new_frame.place(relx=0, rely=0, relwidth=1, relheight=1, y=start_y)

        def animate_step(step):
            if not new_frame.winfo_exists():
                return
            t = step / duration
            progress = ease_out_quint(t)
            current_y = int(start_y * (1 - progress))
            new_frame.place(relx=0, rely=0, relwidth=1, relheight=1, y=current_y)
            if step < duration:
                self.root.after(16, animate_step, step + 1)

        animate_step(0)

    def _batch_rename(self):
        prefix = self.prefix_entry.get().strip()
        if not self.selected_files or not prefix:
            messagebox.showwarning("提示", "请先选择文件并输入有效的前缀！")
            return
        try:
            count = 0
            for idx, file_path in enumerate(self.selected_files, start=1):
                folder, ext = os.path.split(file_path)[0], os.path.splitext(file_path)[1]
                new_path = os.path.join(folder, f"{prefix}_{idx}{ext}")
                os.rename(file_path, new_path)
                count += 1
            messagebox.showinfo("成功", f"已成功重命名 {count} 个文件！")
            self.selected_files.clear()
            self._render_file_items()
        except Exception as e:
            messagebox.showerror("错误", str(e))

    def _add_todo(self):
        text = self.todo_entry.get().strip()
        if text:
            self.pending_todos.append(text)
            self._refresh_todo_lists()
            self.todo_entry.delete(0, tk.END)

    def _mark_completed(self):
        if self.selected_pending_idx is not None and 0 <= self.selected_pending_idx < len(self.pending_todos):
            item = self.pending_todos.pop(self.selected_pending_idx)
            self.completed_todos.append(item)
            self.selected_pending_idx = None
            self._refresh_todo_lists()

    def _revert_completed(self):
        if self.selected_completed_idx is not None and 0 <= self.selected_completed_idx < len(self.completed_todos):
            item = self.completed_todos.pop(self.selected_completed_idx)
            self.pending_todos.append(item)
            self.selected_completed_idx = None
            self._refresh_todo_lists()

    def _delete_pending_todo(self):
        if self.selected_pending_idx is not None and 0 <= self.selected_pending_idx < len(self.pending_todos):
            del self.pending_todos[self.selected_pending_idx]
            self.selected_pending_idx = None
            self._refresh_todo_lists()

    def _delete_completed_todo(self):
        if self.selected_completed_idx is not None and 0 <= self.selected_completed_idx < len(self.completed_todos):
            del self.completed_todos[self.selected_completed_idx]
            self.selected_completed_idx = None
            self._refresh_todo_lists()

    def monitor_clipboard(self):
        try:
            clip = self.root.clipboard_get().strip()
            if clip and clip != self.last_clipboard_text:
                self.last_clipboard_text = clip
                if clip in self.clipboard_history:
                    self.clipboard_history.remove(clip)
                self.clipboard_history.insert(0, clip)
                if len(self.clipboard_history) > 50:
                    self.clipboard_history.pop()

                if getattr(self, "current_page_name", "") == "剪贴板历史":
                    self._refresh_clipboard_cards()
        except tk.TclError:
            pass
        self.root.after(1000, self.monitor_clipboard)

    def _refresh_calc_hist_display(self):
        if hasattr(self, "calc_hist_listbox") and self.calc_hist_listbox.winfo_exists():
            self.calc_hist_listbox.delete(0, tk.END)
            for item in reversed(self.calc_history):
                self.calc_hist_listbox.insert(tk.END, f"  {item}")

    def _on_calc_btn_click(self, char):
        curr = self.calc_display.get()
        if char in ["C", "CE"]:
            self.calc_display.delete(0, tk.END)
        elif char == "←":
            self.calc_display.delete(len(curr) - 1, tk.END)
        elif char == "=":
            try:
                expr = curr.replace("×", "*").replace("÷", "/")
                res = eval(expr)
                res_str = str(res)

                hist_item = f"{curr} = {res_str}"
                self.calc_history.append(hist_item)

                self.calc_display.delete(0, tk.END)
                self.calc_display.insert(tk.END, res_str)
                self._refresh_calc_hist_display()
            except Exception:
                self.calc_display.delete(0, tk.END)
                self.calc_display.insert(tk.END, "Error")
        else:
            if curr == "Error": self.calc_display.delete(0, tk.END)
            self.calc_display.insert(tk.END, char)

    def _fetch_weather_async(self):
        city = self.city_entry.get().strip()
        if not city: return
        for child in self.weather_info_frame.winfo_children():
            child.destroy()

        lbl_load = tk.Label(self.weather_info_frame, text=f"📡 正在同步【{city}】气象数据...",
                            fg=self.colors["accent"], bg=self.colors["entry_bg"], font=self.font_body)
        lbl_load.pack(anchor="nw")

        self.btn_search.config(state="disabled")
        threading.Thread(target=self._query_weather_api, args=(city,), daemon=True).start()

    def _query_weather_api(self, city):
        try:
            url = f"https://wttr.in/{urllib.parse.quote(city)}?format=j1&lang=zh"
            req = urllib.request.Request(url,
                                         headers={"User-Agent": "curl/7.68.0", "Accept-Language": "zh-CN,zh;q=0.9"})
            with urllib.request.urlopen(req, timeout=8) as response:
                data = json.loads(response.read().decode('utf-8'))

            curr = data["current_condition"][0]

            raw_desc = ""
            if "lang_zh-cn" in curr and curr["lang_zh-cn"]:
                raw_desc = curr["lang_zh-cn"][0]["value"]
            elif "lang_zh" in curr and curr["lang_zh"]:
                raw_desc = curr["lang_zh"][0]["value"]
            elif "weatherDesc" in curr and curr["weatherDesc"]:
                raw_desc = curr["weatherDesc"][0]["value"]

            desc = translate_weather(raw_desc)

            info_rows = [
                ("📍", f"目标城市/地区：{city}"),
                ("🌈", f"天气状况：{desc}"),
                ("🌡️", f"当前气温：{curr.get('temp_C', 'N/A')} °C  (体感 {curr.get('FeelsLikeC', 'N/A')} °C)"),
                ("💧", f"相对湿度：{curr.get('humidity', 'N/A')} %"),
                ("💨", f"风速风向：{curr.get('windspeedKmph', 'N/A')} km/h")
            ]

            self.root.after(0, lambda: self._show_weather_res_colorful(info_rows))
        except Exception as e:
            self.root.after(0, lambda: self._show_weather_error(f"❌ 获取失败: {str(e)}"))

    def _show_weather_res_colorful(self, info_rows):
        for child in self.weather_info_frame.winfo_children():
            child.destroy()

        for ic, text in info_rows:
            ic_col = self.icon_colors.get(ic, self.colors["text_primary"])
            row_frame = tk.Frame(self.weather_info_frame, bg=self.colors["entry_bg"])
            row_frame.pack(fill="x", pady=4)

            tk.Label(row_frame, text=ic, fg=ic_col, bg=self.colors["entry_bg"], font=self.font_header, width=3,
                     anchor="w").pack(side="left")
            tk.Label(row_frame, text=text, fg=self.colors["text_primary"], bg=self.colors["entry_bg"],
                     font=self.font_body, anchor="w").pack(side="left")

        self.btn_search.config(state="normal")

    def _show_weather_error(self, err_msg):
